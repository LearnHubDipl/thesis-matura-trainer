' --- Lernfortschritt ---
package "Lernfortschritt & Status" {
  entity QuestionPool {
    * id : Long
  }

  entity QuestionPoolEntry {
    * id : Long
    --
    correctCount : Integer
    lastAnsweredCorrectly : Boolean
    answeredAt : DateTime
  }

  entity StreakTracking {
    * id : Long
    --
    lastLoggedInAt: DateTime
    streak : Integer
  }
}

package "Prüfungsmodul" {
  entity Exam {
    * id : Long
    --
    score : Double
    timeLimit : Integer
    startedAt: DateTime
    finishedAt: DateTime
  }

  entity ExamQuestion {
    * id : Long
    --
    ' selectedAnswers wird jetzt über die Relation zu Answer abgebildet
    freeTextAnswer : String
    isCorrect : Boolean
  }
}

  entity SolutionStep {
    * id : Long

    title : String
    text : String
  }
}
